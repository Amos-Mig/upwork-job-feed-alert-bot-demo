def job_matches(job, cfg):
    hay = (job['title'] + ' ' + job['skills']).lower()
    if not any(k in hay for k in cfg.keywords):
        return False, 'no keyword match'
    if job['budget'] < cfg.min_budget:
        return False, 'budget too low'
    if cfg.verified_only and not job['client_verified']:
        return False, 'client unverified'
    return True, None
